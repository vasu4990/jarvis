Voice Input (Whisper STT)

audio_manager.py

Real-time audio capture with VAD (Voice Activity Detection)

import whisper
import pyaudio
import numpy as np

class AudioManager:
    def __init__(self, model_size="small"):
        self.model = whisper.load_model(model_size)
        self.audio = pyaudio.PyAudio()
        self.stream = None
        
    def start_listening(self):
        self.stream = self.audio.open(
            format=pyaudio.paInt16,
            channels=1,
            rate=16000,
            input=True,
            frames_per_buffer=1024
        )
        
    def transcribe(self, audio_data):
        result = self.model.transcribe(
            audio_data,
            language="en",
            fp16=False
        )
        return result["text"]

planner_cloud.py

GPT-4 powered multi-step task planning

from openai import OpenAI

class CloudPlanner:
    def __init__(self, api_key):
        self.client = OpenAI(api_key=api_key)
        
    def plan(self, user_query, available_tools):
        prompt = f"""
        User request: {user_query}
        Available tools: {available_tools}
        
        Generate a step-by-step plan using the available tools.
        Return JSON format with: [
            {{"tool": "tool_name", "args": {{}}}},
            ...
        ]
        """
        
        response = self.client.chat.completions.create(
            model="gpt-4",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.7
        )
        
        return response.choices[0].message.content

os_powershell.py

PowerShell command execution for Windows automation

import subprocess

class PowerShellTool:
    def execute(self, command, timeout=30):
        try:
            result = subprocess.run(
                ["powershell", "-Command", command],
                capture_output=True,
                text=True,
                timeout=timeout
            )
            return {
                "success": result.returncode == 0,
                "output": result.stdout,
                "error": result.stderr
            }
        except subprocess.TimeoutExpired:
            return {"success": False, "error": "Timeout"}
            
    def launch_app(self, app_name):
        command = f"Start-Process '{app_name}'"
        return self.execute(command)

Vector Memory

vector_store.py

ChromaDB vector storage with semantic search

import chromadb
from sentence_transformers import SentenceTransformer

class VectorStore:
    def __init__(self, path="./data/chroma"):
        self.client = chromadb.PersistentClient(path=path)
        self.collection = self.client.get_or_create_collection("jarvis_memory")
        self.embedder = SentenceTransformer('all-MiniLM-L6-v2')
        
    def store(self, text, metadata=None):
        embedding = self.embedder.encode(text).tolist()
        self.collection.add(
            embeddings=[embedding],
            documents=[text],
            metadatas=[metadata or {}],
            ids=[str(hash(text))]
        )
        
    def search(self, query, n_results=5):
        query_embedding = self.embedder.encode(query).tolist()
        results = self.collection.query(
            query_embeddings=[query_embedding],
            n_results=n_results
        )
        return results["documents"][0]

Safety Layer

policy_gate.py

3-tier permission system (GREEN/YELLOW/RED)

class PolicyGate:
    def __init__(self):
        self.rules = {
            "green": [  # Auto-approved
                "search", "read", "display", "calculate"
            ],
            "yellow": [  # Requires confirmation
                "delete", "modify", "send", "install"
            ],
            "red": [  # Blocked
                "format", "system_shutdown", "registry_edit"
            ]
        }
        
    def check(self, action):
        for permission, actions in self.rules.items():
            if any(keyword in action.lower() for keyword in actions):
                return permission
        return "yellow"  # Default: require confirmation
        
    def approve(self, action):
        permission = self.check(action)
        
        if permission == "green":
            return True
        elif permission == "yellow":
            return self.request_user_approval(action)
        else:  # red
            return False
